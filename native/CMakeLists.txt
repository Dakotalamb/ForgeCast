cmake_minimum_required(VERSION 3.24)
project(forgecast VERSION 0.4.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
find_package(libobs CONFIG REQUIRED)
find_package(obs-frontend-api CONFIG REQUIRED)
find_package(Qt6 REQUIRED COMPONENTS Core Widgets Network)
add_library(forgecast MODULE forgecast.cpp)
target_link_libraries(forgecast PRIVATE OBS::libobs OBS::obs-frontend-api Qt6::Core Qt6::Widgets Qt6::Network)
set_target_properties(forgecast PROPERTIES PREFIX "")
if(WIN32)
  # Per-user binaries are staged under the directory OBS expects beneath ProgramData.
  install(TARGETS forgecast RUNTIME DESTINATION forgecast/bin/64bit LIBRARY DESTINATION forgecast/bin/64bit)
else()
  install(TARGETS forgecast LIBRARY DESTINATION lib/obs-plugins)
endif()
