cmake_minimum_required(VERSION 3.28...3.30)
include("${CMAKE_CURRENT_SOURCE_DIR}/cmake/common/bootstrap.cmake" NO_POLICY_SCOPE)
project(${_name} VERSION ${_version} LANGUAGES C CXX)
include(compilerconfig)
include(defaults)
include(helpers)

add_library(${CMAKE_PROJECT_NAME} MODULE)
find_package(libobs REQUIRED)
find_package(obs-frontend-api REQUIRED)
find_package(Qt6 REQUIRED COMPONENTS Core Widgets Network)
target_link_libraries(${CMAKE_PROJECT_NAME} PRIVATE OBS::libobs OBS::obs-frontend-api Qt6::Core Qt6::Widgets Qt6::Network)
target_sources(${CMAKE_PROJECT_NAME} PRIVATE src/plugin-main.cpp)
set_target_properties(${CMAKE_PROJECT_NAME} PROPERTIES AUTOMOC ON AUTOUIC ON AUTORCC ON CXX_STANDARD 17 CXX_STANDARD_REQUIRED YES)
set_target_properties_plugin(${CMAKE_PROJECT_NAME} PROPERTIES OUTPUT_NAME ${_name})


find_package(Qt6 REQUIRED COMPONENTS Gui)
add_executable(fdgcast-video-qa src/video-qa.cpp)
target_link_libraries(fdgcast-video-qa PRIVATE OBS::libobs Qt6::Core Qt6::Gui)
set_target_properties(fdgcast-video-qa PROPERTIES CXX_STANDARD 17 CXX_STANDARD_REQUIRED YES)

# This module is copied only into the temporary QA runtime, never the installer.
add_library(fdgcast-canvas-qa MODULE src/frontend-qa.cpp)
target_link_libraries(fdgcast-canvas-qa PRIVATE OBS::libobs OBS::obs-frontend-api Qt6::Core Qt6::Gui Qt6::Widgets)
set_target_properties(fdgcast-canvas-qa PROPERTIES CXX_STANDARD 17 CXX_STANDARD_REQUIRED YES PREFIX "")
